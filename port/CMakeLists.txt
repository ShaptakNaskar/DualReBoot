cmake_minimum_required(VERSION 3.22)
project(BeachArm64 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(UNICORN_ARCH arm CACHE STRING "" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BEACH_UNICORN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.cache/unicorn" CACHE PATH "Pinned Unicorn source")
set(BEACH_DECODED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/work/decoded" CACHE PATH "Validated local APK extraction")
if(NOT EXISTS "${BEACH_UNICORN_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Use python3 scripts/build.py --apk /path/to/MyBeachHD.apk --sdk /path/to/Android/Sdk")
endif()
add_subdirectory("${BEACH_UNICORN_DIR}" "${CMAKE_BINARY_DIR}/unicorn")
add_library(beach-runtime STATIC runtime.cpp graphics.cpp)
target_include_directories(beach-runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(beach-runtime PUBLIC unicorn)
target_compile_options(beach-runtime PRIVATE -Wall -Wextra -Werror=return-type)
if(ANDROID)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/engine_bytes.hpp
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/embed.py
      "${BEACH_DECODED_DIR}/lib/armeabi-v7a/libdbgengine.so"
      ${CMAKE_CURRENT_BINARY_DIR}/engine_bytes.hpp
    DEPENDS embed.py "${BEACH_DECODED_DIR}/lib/armeabi-v7a/libdbgengine.so"
    VERBATIM)
  target_sources(beach-runtime PRIVATE fixed_pipeline.cpp)
  target_link_libraries(beach-runtime PUBLIC GLESv2 EGL log)
  add_library(dbgengine SHARED android.cpp ${CMAKE_CURRENT_BINARY_DIR}/engine_bytes.hpp)
  target_include_directories(dbgengine PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(dbgengine PRIVATE beach-runtime android log GLESv2 EGL dl)
  target_link_options(dbgengine PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
else()
  target_compile_definitions(beach-runtime PRIVATE BEACH_HOST_TEST)
  target_link_libraries(beach-runtime PUBLIC dl)
  add_executable(beach-smoke smoke.cpp)
  target_link_libraries(beach-smoke PRIVATE beach-runtime EGL GL)
endif()
